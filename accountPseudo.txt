START

    INPUT currentBalance
    INPUT depositAmount

    newBalance = currentBalance + depositAmount

    OUTPUT "Money deposited successfully"
    OUTPUT "New balance:", newBalance

//withdrawing money
    INPUT currentBalance
    INPUT withdrawalAmount

    IF withdrawalAmount <= currentBalance THEN

        newBalance = currentBalance - withdrawalAmount

        OUTPUT "Withdrawal successful"
        OUTPUT "New balance: R", newBalance

    ELSE

        OUTPUT "Insufficient funds"

    END IF

//spending spikes
